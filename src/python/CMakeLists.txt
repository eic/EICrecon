# Python frontend (see docs/design/python_frontend.md).
#
# Installs the eicrecon Python package to <prefix>/python/eicrecon: pure-Python
# graph definitions and CLI, the _core extension module (JApplication handle,
# units) and the _calorimetry extension module (calorimetry factory types). The
# eicrecon-py launcher goes to <prefix>/bin.

find_package(Python 3.9 COMPONENTS Interpreter Development.Module)
# Only look for pybind11 if Python was found: pybind11's own Python discovery
# would otherwise retry with REQUIRED and abort the configuration
if(Python_FOUND)
  find_package(pybind11 CONFIG QUIET)
  find_package(Boost 1.84 QUIET) # Boost.PFR with field names (header-only)
endif()
if(NOT Python_FOUND
   OR NOT pybind11_FOUND
   OR NOT Boost_FOUND)
  message(
    WARNING
      "Python frontend (eicrecon-py) not built: needs Python development files "
      "(found: ${Python_FOUND}), pybind11 (found: ${pybind11_FOUND}) and Boost >= 1.84 "
      "(found: ${Boost_FOUND}). Set -DEICRECON_PYTHON=OFF to silence this warning."
  )
  return()
endif()

pybind11_add_module(eicrecon_python_core MODULE bindings/core.cc)
set_target_properties(eicrecon_python_core PROPERTIES OUTPUT_NAME "_core")
target_include_directories(eicrecon_python_core
                           PRIVATE ${PROJECT_SOURCE_DIR}/src)
target_include_directories(eicrecon_python_core SYSTEM
                           PRIVATE ${JANA_INCLUDE_DIR})
target_compile_definitions(
  eicrecon_python_core
  PRIVATE "EICRECON_APP_VERSION_STR=\"${EICRECON_VERSION_FULL}\""
          "JANA_VERSION_MAJOR=${JANA_VERSION_MAJOR}"
          "JANA_VERSION_MINOR=${JANA_VERSION_MINOR}"
          "JANA_VERSION_PATCH=${JANA_VERSION_PATCH}")
target_link_libraries(
  eicrecon_python_core
  PRIVATE ${JANA_LIB}
          ROOT::Core
          Boost::headers
          DD4hep::DDCore
          EDM4EIC::edm4eic
          podio::podio
          dd4hep_library
          log_library
          Microsoft.GSL::GSL)

pybind11_add_module(eicrecon_python_calorimetry MODULE bindings/calorimetry.cc)
set_target_properties(eicrecon_python_calorimetry PROPERTIES OUTPUT_NAME
                                                             "_calorimetry")
target_include_directories(eicrecon_python_calorimetry
                           PRIVATE ${PROJECT_SOURCE_DIR}/src)
target_include_directories(eicrecon_python_calorimetry SYSTEM
                           PRIVATE ${JANA_INCLUDE_DIR})
target_link_libraries(
  eicrecon_python_calorimetry
  PRIVATE ${JANA_LIB}
          Boost::headers
          algorithms_calorimetry_library
          log_library
          DD4hep::DDCore
          DD4hep::DDRec
          EDM4EIC::edm4eic
          EDM4HEP::edm4hep
          podio::podio
          spdlog::spdlog)
target_compile_definitions(
  eicrecon_python_calorimetry
  PRIVATE "JANA_VERSION_MAJOR=${JANA_VERSION_MAJOR}"
          "JANA_VERSION_MINOR=${JANA_VERSION_MINOR}"
          "JANA_VERSION_PATCH=${JANA_VERSION_PATCH}")

# One module per C++ plugin with InitPlugin wiring (bindings/plugins/*.cc,
# generated by tools/generate_bindings.py), compiled with that plugin's include
# directories, definitions and link libraries, so that it sees exactly the
# factory types and algorithms the plugin uses.
include(bindings/plugins/plugins.cmake)
# Imported targets are directory-scoped: import the plugins' optional
# dependencies here too
find_package(IRT ${IRT_VERSION_MIN} QUIET)
if(TARGET IRT)
  # same include-directory fix as plugin_add_irt (cmake/jana_plugin.cmake)
  get_target_property(_irt_includes IRT INTERFACE_INCLUDE_DIRECTORIES)
  list(TRANSFORM _irt_includes REPLACE "/IRT$" "")
  list(REMOVE_DUPLICATES _irt_includes)
  set_target_properties(IRT PROPERTIES INTERFACE_INCLUDE_DIRECTORIES
                                       "${_irt_includes}")
endif()
find_package(IRT2 ${IRT2_VERSION_MIN} QUIET)
find_package(
  Boost
  COMPONENTS iostreams
  QUIET)
set(_wiring_targets)
list(LENGTH EICRECON_PYTHON_WIRING_PLUGINS _n)
math(EXPR _last "${_n} - 1")
foreach(_i RANGE 0 ${_last} 2)
  math(EXPR _j "${_i} + 1")
  list(GET EICRECON_PYTHON_WIRING_PLUGINS ${_i} _plugin)
  list(GET EICRECON_PYTHON_WIRING_PLUGINS ${_j} _module)
  if(NOT TARGET ${_plugin}_plugin)
    continue()
  endif()
  set(_target eicrecon_python_wiring_${_module})
  pybind11_add_module(${_target} MODULE bindings/plugins/${_module}.cc)
  set_target_properties(${_target} PROPERTIES OUTPUT_NAME "_wiring_${_module}")
  # Read the plugin's usage at configure time: its directory was processed
  # before this one
  foreach(_property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS LINK_DIRECTORIES
                    LINK_LIBRARIES)
    get_target_property(_${_property} ${_plugin}_plugin ${_property})
    if(NOT _${_property})
      set(_${_property})
    endif()
  endforeach()
  target_include_directories(
    ${_target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/bindings
                       ${_INCLUDE_DIRECTORIES})
  target_compile_definitions(${_target} PRIVATE ${_COMPILE_DEFINITIONS})
  target_link_directories(${_target} PRIVATE ${_LINK_DIRECTORIES})
  target_link_libraries(${_target} PRIVATE ${_LINK_LIBRARIES} Boost::headers)
  list(APPEND _wiring_targets ${_target})
endforeach()

foreach(_target eicrecon_python_core eicrecon_python_calorimetry
                ${_wiring_targets})
  # pybind11_add_module defaults to -fvisibility=hidden. That gives each
  # extension module a private copy of header-defined singletons (e.g.
  # algorithms::GeoSvc::instance(), used in algorithm default member
  # initializers), which are then never initialized by the algorithms_init
  # plugin. Use default visibility, like the JANA plugins.
  set_target_properties(
    ${_target}
    PROPERTIES CXX_VISIBILITY_PRESET default
               VISIBILITY_INLINES_HIDDEN OFF
               INSTALL_RPATH
               "${CMAKE_INSTALL_PREFIX}/${PLUGIN_LIBRARY_OUTPUT_DIRECTORY}")
  install(TARGETS ${_target}
          DESTINATION ${EICRECON_PYTHON_INSTALL_DIR}/eicrecon)
endforeach()

install(
  DIRECTORY eicrecon
  DESTINATION ${EICRECON_PYTHON_INSTALL_DIR}
  FILES_MATCHING
  PATTERN "*.py")
install(PROGRAMS scripts/eicrecon-py DESTINATION ${CMAKE_INSTALL_BINDIR})

# Unit test of the pure-Python graph layer and the command line (no event
# processing). Python itself is not instrumented, so sanitizer builds preload
# the sanitizer runtime.
set(_py_test_env "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/test_package")
set(_py_test_env_modification)
if(USE_ASAN OR USE_UBSAN)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(_sanitizer_runtime_names libclang_rt.asan-x86_64.so)
  else()
    set(_sanitizer_runtime_names libasan.so libubsan.so)
  endif()
  set(_sanitizer_runtimes)
  foreach(_name ${_sanitizer_runtime_names})
    execute_process(
      COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=${_name}
      OUTPUT_VARIABLE _path
      OUTPUT_STRIP_TRAILING_WHITESPACE)
    list(APPEND _sanitizer_runtimes ${_path})
  endforeach()
  list(JOIN _sanitizer_runtimes ":" _sanitizer_preload)
  list(APPEND _py_test_env "LD_PRELOAD=${_sanitizer_preload}")
  # Extend ASAN_OPTIONS from the environment (CI suppressions and options)
  # instead of replacing it:
  #
  # * prepended, so the environment can override it: report ODR violations only
  #   for globals of different sizes (detect_odr_violation=1, as in CI). C++17
  #   inline variables of the standard library (e.g. std::piecewise_construct)
  #   are instrumented in each extension module and would otherwise be reported.
  # * appended, so that they take effect: python3 is not instrumented (no link
  #   order check) and leaks of the interpreter are not of interest here.
  list(APPEND _py_test_env_modification
       "ASAN_OPTIONS=string_prepend:detect_odr_violation=1:"
       "ASAN_OPTIONS=string_append::detect_leaks=0:verify_asan_link_order=0")
endif()
if(NOT USE_TSAN)
  # Assemble an importable package in the build tree: Python sources plus
  # extension modules
  set(_wiring_files)
  foreach(_target ${_wiring_targets})
    list(APPEND _wiring_files $<TARGET_FILE:${_target}>)
  endforeach()
  add_custom_target(
    eicrecon_python_test_package ALL
    COMMAND ${CMAKE_COMMAND} -E make_directory
            ${CMAKE_CURRENT_BINARY_DIR}/test_package/eicrecon
    COMMAND
      ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/eicrecon
      ${CMAKE_CURRENT_BINARY_DIR}/test_package/eicrecon
    COMMAND
      ${CMAKE_COMMAND} -E copy $<TARGET_FILE:eicrecon_python_core>
      $<TARGET_FILE:eicrecon_python_calorimetry> ${_wiring_files}
      ${CMAKE_CURRENT_BINARY_DIR}/test_package/eicrecon
    DEPENDS eicrecon_python_core eicrecon_python_calorimetry ${_wiring_targets}
    COMMENT "Assembling the eicrecon Python package for tests")
  add_test(
    NAME t_python_frontend
    COMMAND ${Python_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_graph.py
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(
    t_python_frontend
    PROPERTIES ENVIRONMENT "${_py_test_env}" ENVIRONMENT_MODIFICATION
               "${_py_test_env_modification}")
endif()
